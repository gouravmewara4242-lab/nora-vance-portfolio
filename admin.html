Supabase SQL Schema

Create `projects` Table in Supabase

Run this one-click SQL query in your Supabase Dashboard → SQL Editor to create the database table: 

-- 1. Create projects table
CREATE TABLE IF NOT EXISTS public.projects (
  id TEXT PRIMARY KEY,
  title TEXT NOT NULL,
  category TEXT NOT NULL,
  category_label TEXT,
  client_name TEXT,
  year TEXT DEFAULT '2026',
  duration TEXT,
  summary TEXT,
  video_url TEXT,
  aspect_ratio TEXT DEFAULT '9:16',
  image_url TEXT,
  featured BOOLEAN DEFAULT true,
  tools TEXT,
  author_email TEXT DEFAULT 'gourav.mewara4242@gmail.com',
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now())
);

-- 2. Enable Row Level Security & Policies
ALTER TABLE public.projects ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Allow public read" ON public.projects;
CREATE POLICY "Allow public read" ON public.projects FOR SELECT TO public USING (true);

DROP POLICY IF EXISTS "Allow insert" ON public.projects;
CREATE POLICY "Allow insert" ON public.projects FOR INSERT TO anon, authenticated WITH CHECK (true);

DROP POLICY IF EXISTS "Allow update" ON public.projects;
CREATE POLICY "Allow update" ON public.projects FOR UPDATE TO anon, authenticated USING (true) WITH CHECK (true);

DROP POLICY IF EXISTS "Allow delete" ON public.projects;
CREATE POLICY "Allow delete" ON public.projects FOR DELETE TO anon, authenticated USING (true);

-- 3. Create Storage bucket 'reels'
INSERT INTO storage.buckets (id, name, public) 
VALUES ('reels', 'reels', true)
ON CONFLICT (id) DO UPDATE SET public = true;

DROP POLICY IF EXISTS "Allow public read reels" ON storage.objects;
CREATE POLICY "Allow public read reels" ON storage.objects FOR SELECT TO public USING (bucket_id = 'reels');

DROP POLICY IF EXISTS "Allow upload reels" ON storage.objects;
CREATE POLICY "Allow upload reels" ON storage.objects FOR INSERT TO anon, authenticated WITH CHECK (bucket_id = 'reels');
Database Deletion SQL

SQL Queries to Delete Videos in Supabase

If your database has Row Level Security enabled, run the DELETE Policy query first. Then use the SQL commands below to delete any video directly from your Supabase SQL Editor: 

-- 1. Ensure Supabase allows DELETE from the Admin panel:
ALTER TABLE public.projects ENABLE ROW LEVEL SECURITY;

DROP POLICY IF EXISTS "Allow delete" ON public.projects;
CREATE POLICY "Allow delete" 
  ON public.projects 
  FOR DELETE 
  TO anon, authenticated 
  USING (true);

-- 2. Delete a specific video by its ID:
DELETE FROM public.projects WHERE id = 'YOUR_PROJECT_ID_HERE';

-- 3. Delete all custom uploaded videos:
DELETE FROM public.projects WHERE id LIKE 'custom-%';

-- 4. Delete all rows from projects table:
DELETE FROM public.projects;
